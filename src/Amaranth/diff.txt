+def inside_circle(x4, y4):
+    """8-gate symmetry check for (2x - 15)^2 + (2y - 15)^2 < 280 on a 16x16 grid."""
+    fx = Mux(x4[3], (~x4[:3]) & 0x7, x4[:3])
+    fy = Mux(y4[3], (~y4[:3]) & 0x7, y4[:3])
+    return (fx + fy) >= 4
